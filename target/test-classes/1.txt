package temperature_sensor_Arch
public
  -- ======================================= 
  -- 1. 数据模型依赖声明
  -- ======================================= 
  annex ISOletteDataModel {**
    data current_temperature
    end current_temperature;
  **};

  -- ======================================= 
  -- 2. 顶层系统容器 
  -- ======================================= 
  system Top_TemperatureSensorSystem
  end Top_TemperatureSensorSystem; 

  system implementation Top_TemperatureSensorSystem.impl
    subcomponents
      tempSensorDevice : device TemperatureSensorDevice.impl;
      tempSensorImpl   : device TemperatureSensorImplementation.impl;
    end Top_TemperatureSensorSystem.impl;

  -- ======================================= 
  -- 3. 温度传感器设备类型定义 
  -- ======================================= 
  device TemperatureSensorDevice
    features
      currentTemperature : out data port isoletteDataModel.current_temperature;
    flows
      current_temperature_flow : flow source currentTemperature;
    end TemperatureSensorDevice;

  device implementation TemperatureSensorDevice.impl
    end TemperatureSensorDevice.impl;

  -- ======================================= 
  -- 4. 温度传感器实体实现类型定义 
  -- ======================================= 
  device TemperatureSensorImplementation
    end TemperatureSensorImplementation;

  device implementation TemperatureSensorImplementation.impl
    end TemperatureSensorImplementation.impl;
end temperature_sensor_Arch;
